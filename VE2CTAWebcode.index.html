    let currentDraggedId = null;

    function handleDragStart(e) {

      currentDraggedId = e.target.id;

      e.dataTransfer.setData('text/plain', e.target.id);

    }

    function handleDragOver(e) {

      e.preventDefault();

      e.currentTarget.classList.add('drop-zone-hover');

    }

    function handleDragLeave(e) {

      e.currentTarget.classList.remove('drop-zone-hover');

    }

    function handleDrop(e, targetCat) {

      e.preventDefault();

      e.currentTarget.classList.remove('drop-zone-hover');

      if (!currentDraggedId) return;

      const draggedEl = document.getElementById(currentDraggedId);

      const expectedCat = draggedEl.dataset.target;

      const targetContent = e.currentTarget.querySelector('.dropped-target-content');

      targetContent.appendChild(draggedEl);

      if (expectedCat === targetCat) {

        draggedEl.className = "p-3 bg-emerald-100 rounded-xl border-2 border-emerald-400 font-bold text-xs text-emerald-900 shadow-sm flex items-center justify-between";

        draggedEl.innerHTML += ` <i class="fa-solid fa-check text-emerald-600 ml-2"></i>`;

        draggedEl.setAttribute('draggable', 'false');

      } else {

        draggedEl.className = "p-3 bg-red-100 rounded-xl border-2 border-red-400 font-bold text-xs text-red-900 shadow-sm";

      }

    }
